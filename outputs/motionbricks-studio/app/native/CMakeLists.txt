cmake_minimum_required(VERSION 3.20)
project(studio_fbx LANGUAGES CXX)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The bundled exporter targets Windows x64 / MSVC")
endif()
set(FBX_SDK_ROOT "" CACHE PATH "Autodesk FBX SDK 2020.3.11 installation/extraction directory")
add_executable(studio_fbx studio_fbx.cpp)
target_compile_features(studio_fbx PRIVATE cxx_std_17)
set_property(TARGET studio_fbx PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_include_directories(studio_fbx PRIVATE "${FBX_SDK_ROOT}/include")
target_link_directories(studio_fbx PRIVATE "${FBX_SDK_ROOT}/lib/x64/release")
target_link_libraries(studio_fbx PRIVATE libfbxsdk-mt libxml2-mt zlib-mt alembic-mt wininet advapi32 ws2_32 bcrypt)
set_target_properties(studio_fbx PROPERTIES RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(studio_fbx_read studio_fbx_read.cpp)
target_compile_features(studio_fbx_read PRIVATE cxx_std_17)
set_property(TARGET studio_fbx_read PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_include_directories(studio_fbx_read PRIVATE "${FBX_SDK_ROOT}/include")
target_link_directories(studio_fbx_read PRIVATE "${FBX_SDK_ROOT}/lib/x64/release")
target_link_libraries(studio_fbx_read PRIVATE libfbxsdk-mt libxml2-mt zlib-mt alembic-mt wininet advapi32 ws2_32 bcrypt)
set_target_properties(studio_fbx_read PROPERTIES RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_CURRENT_SOURCE_DIR}")
